\newcommand{\source}[3]{% x y name : three-state chain of the monitored process
  \node[st] (#3a) at (#1-0.3,#2) {};
  \node[st] (#3b) at (#1,#2) {};
  \node[st] (#3c) at (#1+0.3,#2) {};
  \draw[stlink] (#3a) -- (#3b);
  \draw[stlink] (#3b) -- (#3c);
  \draw[stlink] (#3c.south) to[bend left=65] (#3a.south);
}

  \node[above=0pt of p#3b, yshift=1pt] {$X_#3(t)$};
